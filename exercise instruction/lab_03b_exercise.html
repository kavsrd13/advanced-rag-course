Hierarchical RAG: summary to leaf

Search short document summaries to choose parent documents, then retrieve original detail chunks inside each selected parent.

PowerShell on WindowsPython 3.11 Azure AI Search hierarchy index

What you will learn

The index has two record levels: summary and leaf. A shared documentId links them. Query flow: question → top summaries → ordered parent IDs → leaf search within each parent → answer from leaf text. Summaries are routing aids; original leaf passages are the answer evidence.

5Understand the code

build_index.py

  1. create_index: define vector/text fields plus documentId, level, and chunkOrder.
  2. Leaf pass: read all six PDFs and split them into original detail chunks.
  3. Summary pass: ask the chat model for one routing summary per document, keeping exact IDs and topics.
  4. Parent-child link: summaries and leaves share documentId; the builder embeds both and uploads them to one index.

import os
from pathlib import Path

import pandas as pd
from azure.core.credentials import AzureKeyCredential
from azure.search.documents import SearchClient
from azure.search.documents.indexes import SearchIndexClient
from azure.search.documents.indexes.models import (
    HnswAlgorithmConfiguration,
    SearchField,
    SearchFieldDataType,
    SearchIndex,
    SearchableField,
    SemanticConfiguration,
    SemanticField,
    SemanticPrioritizedFields,
    SemanticSearch,
    SimpleField,
    VectorSearch,
    VectorSearchProfile,
)
from dotenv import load_dotenv
from openai import AzureOpenAI
from pypdf import PdfReader


LAB_FOLDER = Path(__file__).resolve().parent
COURSE_FOLDER = LAB_FOLDER.parent
PDF_FOLDER = COURSE_FOLDER / "Knowledge Base" / "PDF_Files"
METADATA_FILE = COURSE_FOLDER / "Knowledge Base" / "Lab_Support" / "metadata.csv"

# This lab reads credentials from its own .env file.
load_dotenv(LAB_FOLDER / ".env", override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)
search_credential = AzureKeyCredential(os.environ["AZURE_SEARCH_API_KEY"])


def create_index(index_name):
    """Create one index that supports filters, vectors, and semantic ranking."""
    fields = [
        SimpleField(name="id", type=SearchFieldDataType.String, key=True),
        SearchableField(name="title", type=SearchFieldDataType.String),
        SearchableField(name="content", type=SearchFieldDataType.String),
        SearchField(
            name="contentVector",
            type=SearchFieldDataType.Collection(SearchFieldDataType.Single),
            searchable=True,
            vector_search_dimensions=3072,
            vector_search_profile_name="rag-vector-profile",
        ),
        SimpleField(name="source", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="documentId", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="policyId", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="classification", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="product", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="level", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="chunkOrder", type=SearchFieldDataType.Int32, filterable=True),
    ]

    index = SearchIndex(
        name=index_name,
        fields=fields,
        vector_search=VectorSearch(
            algorithms=[HnswAlgorithmConfiguration(name="rag-hnsw")],
            profiles=[
                VectorSearchProfile(
                    name="rag-vector-profile",
                    algorithm_configuration_name="rag-hnsw",
                )
            ],
        ),
        semantic_search=SemanticSearch(
            configurations=[
                SemanticConfiguration(
                    name="rag-semantic-config",
                    prioritized_fields=SemanticPrioritizedFields(
                        title_field=SemanticField(field_name="title"),
                        content_fields=[SemanticField(field_name="content")],
                    ),
                )
            ]
        ),
    )
    SearchIndexClient(
        os.environ["AZURE_SEARCH_ENDPOINT"],
        search_credential,
    ).create_or_update_index(index)


def read_pdf(path):
    """Read page text and keep the page number inside the text."""
    pages = PdfReader(path).pages
    return "\n".join(
        f"[Page {number}] {page.extract_text() or ''}"
        for number, page in enumerate(pages, start=1)
    )


def chunk_text(text, size=1400, overlap=220):
    """Split text into overlapping pieces so boundary information is preserved."""
    chunks = []
    start = 0
    while start < len(text):
        chunks.append(text[start : start + size].strip())
        if start + size >= len(text):
            break
        start += size - overlap
    return chunks


def add_embeddings(documents):
    """Add an embedding vector to every document in small batches."""
    for start in range(0, len(documents), 16):
        batch = documents[start : start + 16]
        response = openai_client.embeddings.create(
            model=os.environ["AZURE_OPENAI_EMBEDDING_DEPLOYMENT"],
            input=[document["content"] for document in batch],
        )
        for document, item in zip(batch, response.data):
            document["contentVector"] = item.embedding


def upload(index_name, documents):
    """Upload documents to Azure AI Search."""
    client = SearchClient(
        os.environ["AZURE_SEARCH_ENDPOINT"],
        index_name,
        search_credential,
    )
    for start in range(0, len(documents), 500):
        client.upload_documents(documents[start : start + 500])


def main():
    hierarchy_index = os.environ["AZURE_SEARCH_HIERARCHY_INDEX"]
    create_index(hierarchy_index)

    # Step 1: Turn every source PDF into searchable leaf chunks.
    leaf_documents = []
    metadata = pd.read_csv(METADATA_FILE)
    for _, row in metadata.iterrows():
        document_id = str(row["sourceFile"])
        source_path = PDF_FOLDER / f"{document_id}.pdf"
        chunks = chunk_text(read_pdf(source_path))

        for order, chunk in enumerate(chunks):
            leaf_documents.append(
                {
                    "id": f"{document_id}-leaf-{order}",
                    "title": str(row["title"]),
                    "content": chunk,
                    "source": source_path.name,
                    "documentId": document_id,
                    "policyId": str(row["documentId"]),
                    "classification": str(row["classification"]),
                    "product": "" if pd.isna(row["product"]) else str(row["product"]),
                    "level": "leaf",
                    "chunkOrder": order,
                }
            )

    # Step 2: Create one model-generated summary for each source document.
    summary_documents = []
    for document_id in metadata["sourceFile"]:
        document_chunks = [
            item for item in leaf_documents if item["documentId"] == document_id
        ]
        # Summarize the original PDF text, not overlapping chunks with repeated lines.
        source_text = read_pdf(PDF_FOLDER / f"{document_id}.pdf")
        response = openai_client.chat.completions.create(
            model=os.environ["AZURE_OPENAI_DEPLOYMENT"],
            messages=[
                {
                    "role": "system",
                    "content": (
                        "Write a short routing summary of this document. Include its main "
                        "topics, exact product and incident IDs, important outcomes, "
                        "and references to other documents. Use only the source text."
                    ),
                },
                {"role": "user", "content": source_text},
            ],
        )
        first = document_chunks[0]
        summary_documents.append(
            {
                **first,
                "id": f"{document_id}-summary",
                "content": response.choices[0].message.content,
                "level": "summary",
                "chunkOrder": 0,
            }
        )

    # Step 3: Put summaries and their detail chunks in the same index.
    add_embeddings(leaf_documents)
    add_embeddings(summary_documents)
    upload(hierarchy_index, leaf_documents + summary_documents)

    print(
        "HIERARCHY INDEX:",
        hierarchy_index,
        "| SUMMARIES:", len(summary_documents),
        "| DETAIL CHUNKS:", len(leaf_documents),
    )


if __name__ == "__main__":
    main()

rag_app.py

  1. Question vector: embed the question once for both retrieval stages.
  2. Stage 1: search only level eq 'summary' and keep ranked parent IDs.
  3. Stage 2: for each chosen parent, search level eq 'leaf' within its documentId.
  4. Generation: put only leaf chunks in the final prompt; print summary matches, selected parents, details, and answer.

import os
import sys
from pathlib import Path

from azure.core.credentials import AzureKeyCredential
from azure.search.documents import SearchClient
from azure.search.documents.models import VectorizedQuery
from dotenv import load_dotenv
from openai import AzureOpenAI


load_dotenv(Path(__file__).with_name(".env"), override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)
search_client = SearchClient(
    os.environ["AZURE_SEARCH_ENDPOINT"],
    os.environ["AZURE_SEARCH_HIERARCHY_INDEX"],
    AzureKeyCredential(os.environ["AZURE_SEARCH_API_KEY"]),
)


def main():
    question = " ".join(sys.argv[1:]) or (
        "Why did X200 E-4012 incidents fall from April to June, "
        "and what should support do if E-4012 occurs again?"
    )
    embedding = openai_client.embeddings.create(
        model=os.environ["AZURE_OPENAI_EMBEDDING_DEPLOYMENT"],
        input=[question],
    ).data[0].embedding
    vector_query = VectorizedQuery(
        vector=embedding,
        k_nearest_neighbors=5,
        fields="contentVector",
    )

    # First retrieve small document summaries to locate relevant documents.
    summary_results = search_client.search(
        search_text=question,
        vector_queries=[vector_query],
        query_type="semantic",
        semantic_configuration_name="rag-semantic-config",
        filter="level eq 'summary'",
        top=3,
        select=["policyId", "documentId", "content", "source", "level"],
    )
    summaries = [dict(result) for result in summary_results]
    # Keep search ranking: a set would scramble which parent documents we open.
    document_ids = list(dict.fromkeys(item["documentId"] for item in summaries))

    # Search each selected document separately so each parent contributes details.
    details = []
    for document_id in document_ids:
        detail_results = search_client.search(
            search_text=question,
            vector_queries=[vector_query],
            query_type="semantic",
            semantic_configuration_name="rag-semantic-config",
            filter=f"documentId eq '{document_id}' and level eq 'leaf'",
            top=3,
            select=["policyId", "documentId", "content", "source", "level"],
        )
        details.extend(dict(result) for result in detail_results)

    # Summaries route the search; original detail chunks support the final answer.
    context = "\n\n".join(
        f"[{item['policyId']}] {item['content']}" for item in details
    )
    response = openai_client.chat.completions.create(
        model=os.environ["AZURE_OPENAI_DEPLOYMENT"],
        messages=[
            {
                "role": "system",
                "content": (
                    "Answer only from the detail evidence. Cite the exact bracketed "
                    "policy IDs shown in the evidence, such as [MAN-X200] or "
                    "[OPS-Q2-2026]. Never write the placeholder [POLICY_ID]. "
                    "If evidence is missing, say so. Use ASCII punctuation."
                ),
            },
            {"role": "user", "content": f"Question: {question}\n\nEvidence:\n{context}"},
        ],
    )

    print("SUMMARY MATCHES:", [item["policyId"] for item in summaries])
    print("SELECTED DOCUMENTS:", document_ids)
    print("DETAIL CHUNKS:", [item["policyId"] for item in details])
    answer = response.choices[0].message.content.translate(
        str.maketrans({"→": "->", "–": "-", "—": "-", "≈": "about "})
    ).encode(
        "ascii", errors="replace"
    ).decode("ascii")
    print("\nANSWER\n", answer)


if __name__ == "__main__":
    main()
