What you will learn

The route is PDF → chunks → embeddings → local JSON → question embedding → cosine similarity → top passages → answer. Cosine similarity is (q · d) / (||q|| ||d||); larger values indicate more similar vector directions. This is a local vector store demo, not an Azure AI Search index.

5Understand the code

build_index.py

  1. Read inputs: load each PDF and its policy metadata from the CSV.
  2. chunk_text: split page text with overlap so facts near a boundary remain retrievable.
  3. add_embeddings: generate a vector per chunk in batches of 16.
  4. write_local_index: save text, metadata, and vectors to local_vector_index.json; no Search upload occurs.
Read the complete embedded build_index.py source
import json
import os
from pathlib import Path

import pandas as pd
from dotenv import load_dotenv
from openai import AzureOpenAI
from pypdf import PdfReader


LAB_FOLDER = Path(__file__).resolve().parent
COURSE_FOLDER = LAB_FOLDER.parent
PDF_FOLDER = COURSE_FOLDER / "Knowledge Base" / "PDF_Files"
METADATA_FILE = COURSE_FOLDER / "Knowledge Base" / "Lab_Support" / "metadata.csv"
LOCAL_INDEX_FILE = LAB_FOLDER / "local_vector_index.json"

# This lab reads credentials from its own .env file.
load_dotenv(LAB_FOLDER / ".env", override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)


def read_pdf(path):
    """Read page text and keep the page number inside the text."""
    pages = PdfReader(path).pages
    return "\n".join(
        f"[Page {number}] {page.extract_text() or ''}"
        for number, page in enumerate(pages, start=1)
    )


def chunk_text(text, size=1400, overlap=220):
    """Split text into overlapping pieces so boundary information is preserved."""
    chunks = []
    start = 0
    while start < len(text):
        chunks.append(text[start : start + size].strip())
        if start + size >= len(text):
            break
        start += size - overlap
    return chunks


def add_embeddings(documents):
    """Add an embedding vector to every document in small batches."""
    for start in range(0, len(documents), 16):
        batch = documents[start : start + 16]
        response = openai_client.embeddings.create(
            model=os.environ["AZURE_OPENAI_EMBEDDING_DEPLOYMENT"],
            input=[document["content"] for document in batch],
        )
        for document, item in zip(batch, response.data):
            document["contentVector"] = item.embedding


def write_local_index(documents):
    with LOCAL_INDEX_FILE.open("w", encoding="utf-8") as file_handle:
        json.dump(documents, file_handle, ensure_ascii=False, indent=2)


def main():
    documents = []
    metadata = pd.read_csv(METADATA_FILE)
    for _, row in metadata.iterrows():
        document_id = str(row["sourceFile"])
        source_path = PDF_FOLDER / f"{document_id}.pdf"
        chunks = chunk_text(read_pdf(source_path))

        for order, chunk in enumerate(chunks):
            documents.append(
                {
                    "id": f"{document_id}-leaf-{order}",
                    "title": str(row["title"]),
                    "content": chunk,
                    "source": source_path.name,
                    "documentId": document_id,
                    "policyId": str(row["documentId"]),
                    "owner": str(row["owner"]),
                    "version": str(row["version"]),
                    "effectiveDate": str(row["effectiveDate"]),
                    "classification": str(row["classification"]),
                    "status": str(row["status"]),
                    "product": "" if pd.isna(row["product"]) else str(row["product"]),
                }
            )

    add_embeddings(documents)
    write_local_index(documents)
    print("LOCAL INDEX:", LOCAL_INDEX_FILE)
    print("CHUNKS:", len(documents))


if __name__ == "__main__":
    main()

rag_app.py

  1. load_local_index: read the JSON created by the builder.
  2. cosine_similarity: calculate dot product divided by vector lengths.
  3. retrieve_semantic: embed the question, score every chunk, and keep the highest-scoring twelve.
  4. answer and main: ground the chat model in those chunks and print scores, sources, and answer.
Read the complete embedded rag_app.py source
import json
import math
import os
from pathlib import Path

from dotenv import load_dotenv
from openai import AzureOpenAI


LAB_FOLDER = Path(__file__).resolve().parent
LOCAL_INDEX_FILE = LAB_FOLDER / "local_vector_index.json"

# This lab reads credentials from its own .env file.
load_dotenv(LAB_FOLDER / ".env", override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)

DEFAULT_TOP_K = 12


def load_local_index():
    if not LOCAL_INDEX_FILE.exists():
        raise FileNotFoundError(
            f"Local index not found: {LOCAL_INDEX_FILE}. "
            "Run build_index.py first."
        )
    with LOCAL_INDEX_FILE.open("r", encoding="utf-8") as file_handle:
        return json.load(file_handle)


def cosine_similarity(vector_a, vector_b):
    dot = sum(a * b for a, b in zip(vector_a, vector_b))
    norm_a = math.sqrt(sum(a * a for a in vector_a))
    norm_b = math.sqrt(sum(b * b for b in vector_b))
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return dot / (norm_a * norm_b)


def retrieve_semantic(question, documents, top_k=DEFAULT_TOP_K):
    question_vector = openai_client.embeddings.create(
        model=os.environ["AZURE_OPENAI_EMBEDDING_DEPLOYMENT"],
        input=[question],
    ).data[0].embedding

    scored = []
    for document in documents:
        score = cosine_similarity(question_vector, document["contentVector"])
        scored.append((score, document))

    scored.sort(key=lambda item: item[0], reverse=True)
    return scored[:top_k]


def answer(question, scored_chunks):
    context = "\n\n".join(
        f"[{chunk['policyId']}] {chunk['content']}" for _, chunk in scored_chunks
    )
    response = openai_client.chat.completions.create(
        model=os.environ["AZURE_OPENAI_DEPLOYMENT"],
        messages=[
            {
                "role": "system",
                "content": "Answer only from the context. Cite actual policy IDs in brackets, such as [SEC-114]. If evidence is missing, say so.",
            },
            {
                "role": "user",
                "content": f"Question: {question}\n\nContext:\n{context}",
            },
        ],
    )
    return response.choices[0].message.content


def main():
    question = "What does SEC-114 require for laptops?"
    documents = load_local_index()
    scored_chunks = retrieve_semantic(question, documents, top_k=DEFAULT_TOP_K)
    response = answer(question, scored_chunks)

    print("QUESTION\n", question)
    print("\nRETRIEVED CHUNKS")
    for score, chunk in scored_chunks:
        print(f"- score={score:.4f} [{chunk['policyId']}] {chunk['source']}")
    print("\nANSWER\n", response)


if __name__ == "__main__":
    main()

import gradio as gr

from rag_app import answer, load_local_index, retrieve_semantic


SUGGESTED_PROMPTS = [
    "What does SEC-114 require for laptops?",
    "Summarize remote work data handling expectations for employees.",
    "How is error E-4012 resolved for ContosoPrint X200?",
    "Find every page across all six documents where the exact phrase 'ContosoPrint X200' appears, and give a full per-document count.",
    "List every policy effective after 2026-08-01 with version and owner in a table.",
]


def run_rag(question):
    if not question.strip():
        raise gr.Error("Please enter a question.")

    documents = load_local_index()
    scored_chunks = retrieve_semantic(question, documents, top_k=12)
    response = answer(question, scored_chunks)
    return f"### Answer\n\n{response}"


def build_demo():
    with gr.Blocks(title="Semantic only RAG - Visual Tester") as demo:
        gr.Markdown("# Semantic only RAG: Visual Tester")
        question = gr.Textbox(
            label="Question",
            lines=4,
            placeholder="Ask a question...",
        )
        run_button = gr.Button("Run RAG", variant="primary")
        output = gr.Markdown()

        with gr.Accordion("Suggested prompts", open=False):
            gr.Examples(
                examples=[[item] for item in SUGGESTED_PROMPTS],
                inputs=question,
                label="Click a prompt to copy it into the question box",
            )

        run_button.click(run_rag, inputs=question, outputs=output)
        question.submit(run_rag, inputs=question, outputs=output)

    return demo


if __name__ == "__main__":
    build_demo().launch()
