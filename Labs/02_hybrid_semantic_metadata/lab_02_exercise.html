Hybrid retrieval and metadata filtering

Combine exact-word search and vector similarity, filter to X200 documents, and inspect the scores with semantic reranking off.

What you will learn

The builder stores text, vectors, and metadata. At query time the app runs keyword search, vector-only search, then a hybrid query within product eq 'ContosoPrint X200'. The model sees the first six hybrid candidates. The score columns are diagnostic: keyword, vector, and hybrid scores come from different ranking methods and should not be compared as if they share a scale.

build_index.py

  1. Configuration: load this lab’s .env, PDF folder, and metadata.csv.
  2. create_index: define searchable text, a 3,072-dimension vector, filterable metadata, and rag-semantic-config.
  3. read_pdf and chunk_text: extract page text and create overlapping chunks.
  4. add_embeddings and upload: call the embedding deployment and upload chunks to Azure AI Search.

  6. main: include product metadata, so the app can constrain retrieval before ranking.

import os
from pathlib import Path

import pandas as pd
from azure.core.credentials import AzureKeyCredential
from azure.search.documents import SearchClient
from azure.search.documents.indexes import SearchIndexClient
from azure.search.documents.indexes.models import (
    HnswAlgorithmConfiguration,
    SearchField,
    SearchFieldDataType,
    SearchIndex,
    SearchableField,
    SemanticConfiguration,
    SemanticField,
    SemanticPrioritizedFields,
    SemanticSearch,
    SimpleField,
    VectorSearch,
    VectorSearchProfile,
)
from dotenv import load_dotenv
from openai import AzureOpenAI
from pypdf import PdfReader


LAB_FOLDER = Path(__file__).resolve().parent
COURSE_FOLDER = LAB_FOLDER.parent
PDF_FOLDER = COURSE_FOLDER / "Knowledge Base" / "PDF_Files"
METADATA_FILE = COURSE_FOLDER / "Knowledge Base" / "Lab_Support" / "metadata.csv"

# This lab reads credentials from its own .env file.
load_dotenv(LAB_FOLDER / ".env", override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)
search_credential = AzureKeyCredential(os.environ["AZURE_SEARCH_API_KEY"])


def create_index(index_name):
    """Create one index that supports filters, vectors, and semantic ranking."""
    fields = [
        SimpleField(name="id", type=SearchFieldDataType.String, key=True),
        SearchableField(name="title", type=SearchFieldDataType.String),
        SearchableField(name="content", type=SearchFieldDataType.String),
        SearchField(
            name="contentVector",
            type=SearchFieldDataType.Collection(SearchFieldDataType.Single),
            searchable=True,
            vector_search_dimensions=3072,
            vector_search_profile_name="rag-vector-profile",
        ),
        SimpleField(name="source", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="documentId", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="policyId", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="classification", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="product", type=SearchFieldDataType.String, filterable=True),
    ]

    index = SearchIndex(
        name=index_name,
        fields=fields,
        vector_search=VectorSearch(
            algorithms=[HnswAlgorithmConfiguration(name="rag-hnsw")],
            profiles=[
                VectorSearchProfile(
                    name="rag-vector-profile",
                    algorithm_configuration_name="rag-hnsw",
                )
            ],
        ),
        semantic_search=SemanticSearch(
            configurations=[
                SemanticConfiguration(
                    name="rag-semantic-config",
                    prioritized_fields=SemanticPrioritizedFields(
                        title_field=SemanticField(field_name="title"),
                        content_fields=[SemanticField(field_name="content")],
                    ),
                )
            ]
        ),
    )
    SearchIndexClient(
        os.environ["AZURE_SEARCH_ENDPOINT"],
        search_credential,
    ).create_or_update_index(index)


def read_pdf(path):
    """Read page text and keep the page number inside the text."""
    pages = PdfReader(path).pages
    return "\n".join(
        f"[Page {number}] {page.extract_text() or ''}"
        for number, page in enumerate(pages, start=1)
    )


def chunk_text(text, size=1400, overlap=220):
    """Split text into overlapping pieces so boundary information is preserved."""
    chunks = []
    start = 0
    while start < len(text):
        chunks.append(text[start : start + size].strip())
        if start + size >= len(text):
            break
        start += size - overlap
    return chunks


def add_embeddings(documents):
    """Add an embedding vector to every document in small batches."""
    for start in range(0, len(documents), 16):
        batch = documents[start : start + 16]
        response = openai_client.embeddings.create(
            model=os.environ["AZURE_OPENAI_EMBEDDING_DEPLOYMENT"],
            input=[document["content"] for document in batch],
        )
        for document, item in zip(batch, response.data):
            document["contentVector"] = item.embedding


def upload(index_name, documents):
    """Upload documents to Azure AI Search."""
    client = SearchClient(
        os.environ["AZURE_SEARCH_ENDPOINT"],
        index_name,
        search_credential,
    )
    for start in range(0, len(documents), 500):
        client.upload_documents(documents[start : start + 500])


def main():
    base_index = os.environ["AZURE_SEARCH_INDEX"]
    create_index(base_index)

    # Step 1: Turn every source PDF into searchable leaf chunks.
    leaf_documents = []
    metadata = pd.read_csv(METADATA_FILE)
    for _, row in metadata.iterrows():
        document_id = str(row["sourceFile"])
        source_path = PDF_FOLDER / f"{document_id}.pdf"
        chunks = chunk_text(read_pdf(source_path))

        for order, chunk in enumerate(chunks):
            leaf_documents.append(
                {
                    "id": f"{document_id}-leaf-{order}",
                    "title": str(row["title"]),
                    "content": chunk,
                    "source": source_path.name,
                    "documentId": document_id,
                    "policyId": str(row["documentId"]),
                    "classification": str(row["classification"]),
                    "product": "" if pd.isna(row["product"]) else str(row["product"]),
                }
            )

    # Step 2: Embed the chunks and upload them to this lesson's index.
    add_embeddings(leaf_documents)
    upload(base_index, leaf_documents)
    print("INDEX:", base_index, "| CHUNKS:", len(leaf_documents))


if __name__ == "__main__":
    main()

rag_app.py

  1. answer: send only selected candidate passages to the chat deployment.
  2. retrieve_hybrid: embed the question, run keyword and vector diagnostics, then combine keyword plus vector with the same X200 filter.
  3. Candidate selection: keep the first six hybrid results; semantic reranking is not requested in this lab.
  4. main: ask a fixed E-4012 question and print candidate counts and answer.

import os
from pathlib import Path

from azure.core.credentials import AzureKeyCredential
from azure.search.documents import SearchClient
from azure.search.documents.models import VectorizedQuery
from dotenv import load_dotenv
from openai import AzureOpenAI


load_dotenv(Path(__file__).with_name(".env"), override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)
search_client = SearchClient(
    os.environ["AZURE_SEARCH_ENDPOINT"],
    os.environ["AZURE_SEARCH_INDEX"],
    AzureKeyCredential(os.environ["AZURE_SEARCH_API_KEY"]),
)


def answer(question, chunks):
    context = "\n\n".join(
        f"[{chunk['policyId']}] {chunk['content']}" for chunk in chunks
    )
    response = openai_client.chat.completions.create(
        model=os.environ["AZURE_OPENAI_DEPLOYMENT"],
        messages=[
            {"role": "system", "content": "Answer only from the evidence. Cite actual policy IDs in brackets, such as [MAN-X200]. If evidence is missing, say so."},
            {"role": "user", "content": f"Question: {question}\n\nEvidence:\n{context}"},
        ],
    )
    return response.choices[0].message.content


def retrieve_hybrid(question):
    embedding = openai_client.embeddings.create(
        model=os.environ["AZURE_OPENAI_EMBEDDING_DEPLOYMENT"],
        input=[question],
    ).data[0].embedding

    keyword_results = search_client.search(
        search_text=question,
        filter="product eq 'ContosoPrint X200'",
        top=50,
        select=["id"],
    )
    keyword_scores = {
        item["id"]: item.get("@search.score") for item in keyword_results if item.get("id")
    }

    vector_results = search_client.search(
        search_text="",
        vector_queries=[
            VectorizedQuery(vector=embedding, k_nearest_neighbors=50, fields="contentVector")
        ],
        filter="product eq 'ContosoPrint X200'",
        top=50,
        select=["id"],
    )
    vector_scores = {
        item["id"]: item.get("@search.score") for item in vector_results if item.get("id")
    }

    results = search_client.search(
        search_text=question,
        vector_queries=[
            VectorizedQuery(vector=embedding, k_nearest_neighbors=12, fields="contentVector")
        ],
        filter="product eq 'ContosoPrint X200'",
        top=12,
        select=["id", "policyId", "title", "content", "source", "product"],
    )
    candidates = [dict(result) for result in results]
    for candidate in candidates:
        candidate["keyword_score"] = keyword_scores.get(candidate["id"])
        candidate["vector_score"] = vector_scores.get(candidate["id"])
        candidate["hybrid_score"] = candidate.get("@search.score")
        candidate["reranker_score"] = candidate.get("@search.reranker_score")

    selected = candidates[:6]
    return candidates, selected


def main():
    question = "How is error E-4012 resolved?"

    # Hybrid retrieval combines keyword + vector without semantic reranker.
    candidates, selected = retrieve_hybrid(question)

    print("CANDIDATES RETRIEVED:", len(candidates))
    print("CHUNKS AFTER SELECTION:", len(selected))
    response = answer(question, selected).encode("ascii", errors="replace").decode("ascii")
    print("\nANSWER\n", response)


if __name__ == "__main__":
    main()

import gradio as gr

from rag_app import answer, retrieve_hybrid


SUGGESTED_PROMPTS = [
    "How is error E-4012 resolved?",
    "What were the key actions from the Q2 operations incident review?",
    "What governance controls are required for RAG security evaluations?",
]


def format_scores(reranked):
    lines = ["### Hybrid scores (reranker OFF)"]
    for index, chunk in enumerate(reranked, start=1):
        keyword_score = chunk.get("keyword_score")
        vector_score = chunk.get("vector_score")
        hybrid_score = chunk.get("hybrid_score")
        reranker_score = chunk.get("reranker_score")
        keyword_text = "n/a" if keyword_score is None else f"{keyword_score:.4f}"
        vector_text = "n/a" if vector_score is None else f"{vector_score:.4f}"
        hybrid_text = "n/a" if hybrid_score is None else f"{hybrid_score:.4f}"
        reranker_text = "n/a" if reranker_score is None else f"{reranker_score:.4f}"
        lines.append(
            f"{index}. [{chunk['policyId']}] {chunk['source']}  \n"
            f"keyword={keyword_text}, vector={vector_text}, hybrid={hybrid_text}, reranker={reranker_text}"
        )
    return "\n\n".join(lines)


def run_rag(question):
    if not question.strip():
        raise gr.Error("Please enter a question.")

    _, reranked = retrieve_hybrid(question)
    response = answer(question, reranked)
    return f"### Answer\n\n{response}", format_scores(reranked)


def build_demo():
    with gr.Blocks(title="Lab 02 - Hybrid RAG Visual Tester (reranker OFF)") as demo:
        gr.Markdown("# Lab 02: Hybrid RAG Visual Tester (reranker OFF)")
        question = gr.Textbox(
            label="Question",
            lines=4,
            placeholder="Ask a question...",
        )
        run_button = gr.Button("Run RAG", variant="primary")
        with gr.Row():
            with gr.Column(scale=3):
                output = gr.Markdown()
            with gr.Column(scale=2):
                scores = gr.Markdown()

        with gr.Accordion("Suggested prompts", open=False):
            gr.Examples(
                examples=[[item] for item in SUGGESTED_PROMPTS],
                inputs=question,
                label="Click a prompt to copy it into the question box",
            )

        run_button.click(run_rag, inputs=question, outputs=[output, scores])
        question.submit(run_rag, inputs=question, outputs=[output, scores])

    return demo


if __name__ == "__main__":
    build_demo().launch()
