Builder: build_index.py

  1. Imports and configuration: read this lab’s .env and locate the six PDFs plus metadata.csv.
  2. create_index: define text fields, a 3,072-number vector field, document labels, and the filterable allowedUsers collection. The vector is stored for later RAG patterns; this lab’s question path uses keyword search.
  3. read_pdf and chunk_text: extract page text and split it into overlapping passages of about 1,400 characters.
  4. main: join each PDF to its metadata row. Every chunk inherits its document ID, classification, and allowed users.
  5. add_embeddings and upload: call the embedding deployment in batches, then send chunks to the Search index. Look for INDEX: ... | CHUNKS: ....

import os
from pathlib import Path

import pandas as pd
from azure.core.credentials import AzureKeyCredential
from azure.search.documents import SearchClient
from azure.search.documents.indexes import SearchIndexClient
from azure.search.documents.indexes.models import (
    HnswAlgorithmConfiguration,
    SearchField,
    SearchFieldDataType,
    SearchIndex,
    SearchableField,
    SimpleField,
    VectorSearch,
    VectorSearchProfile,
)
from dotenv import load_dotenv
from openai import AzureOpenAI
from pypdf import PdfReader


LAB_FOLDER = Path(__file__).resolve().parent
COURSE_FOLDER = LAB_FOLDER.parent
PDF_FOLDER = COURSE_FOLDER / "Knowledge Base" / "PDF_Files"
METADATA_FILE = COURSE_FOLDER / "Knowledge Base" / "Lab_Support" / "metadata.csv"

# This lab reads credentials from its own .env file.
load_dotenv(LAB_FOLDER / ".env", override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)
search_credential = AzureKeyCredential(os.environ["AZURE_SEARCH_API_KEY"])


def create_index(index_name):
    """Create one index that stores text, vectors, and access metadata."""
    fields = [
        SimpleField(name="id", type=SearchFieldDataType.String, key=True),
        SearchableField(name="title", type=SearchFieldDataType.String),
        SearchableField(name="content", type=SearchFieldDataType.String),
        SearchField(
            name="contentVector",
            type=SearchFieldDataType.Collection(SearchFieldDataType.Single),
            searchable=True,
            vector_search_dimensions=3072,
            vector_search_profile_name="rag-vector-profile",
        ),
        SimpleField(name="source", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="documentId", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="policyId", type=SearchFieldDataType.String, filterable=True),
        SimpleField(name="classification", type=SearchFieldDataType.String, filterable=True),
        SearchField(
            name="allowedUsers",
            type=SearchFieldDataType.Collection(SearchFieldDataType.String),
            filterable=True,
            hidden=True,
        ),
    ]

    index = SearchIndex(
        name=index_name,
        fields=fields,
        vector_search=VectorSearch(
            algorithms=[HnswAlgorithmConfiguration(name="rag-hnsw")],
            profiles=[
                VectorSearchProfile(
                    name="rag-vector-profile",
                    algorithm_configuration_name="rag-hnsw",
                )
            ],
        ),

    )
    SearchIndexClient(
        os.environ["AZURE_SEARCH_ENDPOINT"],
        search_credential,
    ).create_or_update_index(index)


def read_pdf(path):
    """Read page text and keep the page number inside the text."""
    pages = PdfReader(path).pages
    return "\n".join(
        f"[Page {number}] {page.extract_text() or ''}"
        for number, page in enumerate(pages, start=1)
    )


def chunk_text(text, size=1400, overlap=220):
    """Split text into overlapping pieces so boundary information is preserved."""
    chunks = []
    start = 0
    while start < len(text):
        chunks.append(text[start : start + size].strip())
        if start + size >= len(text):
            break
        start += size - overlap
    return chunks


def add_embeddings(documents):
    """Add an embedding vector to every document in small batches."""
    for start in range(0, len(documents), 16):
        batch = documents[start : start + 16]
        response = openai_client.embeddings.create(
            model=os.environ["AZURE_OPENAI_EMBEDDING_DEPLOYMENT"],
            input=[document["content"] for document in batch],
        )
        for document, item in zip(batch, response.data):
            document["contentVector"] = item.embedding


def upload(index_name, documents):
    """Upload documents to Azure AI Search."""
    client = SearchClient(
        os.environ["AZURE_SEARCH_ENDPOINT"],
        index_name,
        search_credential,
    )
    for start in range(0, len(documents), 500):
        client.upload_documents(documents[start : start + 500])


def main():
    base_index = os.environ["AZURE_SEARCH_INDEX"]
    create_index(base_index)

    # Step 1: Turn every source PDF into searchable leaf chunks.
    leaf_documents = []
    metadata = pd.read_csv(METADATA_FILE)
    for _, row in metadata.iterrows():
        document_id = str(row["sourceFile"])
        source_path = PDF_FOLDER / f"{document_id}.pdf"
        chunks = chunk_text(read_pdf(source_path))

        for order, chunk in enumerate(chunks):
            leaf_documents.append(
                {
                    "id": f"{document_id}-leaf-{order}",
                    "title": str(row["title"]),
                    "content": chunk,
                    "source": source_path.name,
                    "documentId": document_id,
                    "policyId": str(row["documentId"]),
                    "classification": str(row["classification"]),
                    "allowedUsers": str(row["allowedUsers"]).split("|"),
                }
            )

    # Step 2: Embed the chunks and upload them to this lesson's index.
    add_embeddings(leaf_documents)
    upload(base_index, leaf_documents)
    print("INDEX:", base_index, "| CHUNKS:", len(leaf_documents))


if __name__ == "__main__":
    main()

Console pipeline: rag_app.py

  1. access_filter: construct allowedUsers/any(...) for all plus the selected user.
  2. retrieve_chunks: pass that filter into Azure AI Search with search_text. Filtering happens before the top three candidates are returned.
  3. generate_answer: make a context string from those authorized candidates only. If there are none, return a fixed no-answer message.
  4. main: run a fixed question as Alice and print the filter, retrieved source IDs, and answer.

import os
from pathlib import Path

from azure.core.credentials import AzureKeyCredential
from azure.search.documents import SearchClient
from dotenv import load_dotenv
from openai import AzureOpenAI


# Each lab reads credentials from the .env file in its own folder.
load_dotenv(Path(__file__).with_name(".env"), override=True)

openai_client = AzureOpenAI(
    api_key=os.environ["AZURE_OPENAI_API_KEY"],
    azure_endpoint=os.environ["AZURE_OPENAI_ENDPOINT"].replace("/openai/v1", ""),
    api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-06-01"),
)
search_client = SearchClient(
    os.environ["AZURE_SEARCH_ENDPOINT"],
    os.environ["AZURE_SEARCH_INDEX"],
    AzureKeyCredential(os.environ["AZURE_SEARCH_API_KEY"]),
)


def access_filter(user_id):
    """Allow public documents plus documents assigned to this classroom user."""
    return f"allowedUsers/any(u: search.in(u, 'all,{user_id}', ','))"


def retrieve_chunks(question, user_id, top_k=3):
    """Run keyword retrieval only inside the signed-in user's allowed content."""
    results = search_client.search(
        search_text=question,
        filter=access_filter(user_id),
        top=top_k,
        select=["policyId", "title", "content", "source", "classification"],
    )
    return [dict(result) for result in results]


def generate_answer(question, chunks):
    """Give only authorized retrieved chunks to the answer model."""
    if not chunks:
        return "No authorized document contains this answer."

    context = "\n\n".join(
        f"[{chunk['policyId']}] {chunk['content']}" for chunk in chunks
    )
    response = openai_client.chat.completions.create(
        model=os.environ["AZURE_OPENAI_DEPLOYMENT"],
        messages=[
            {
                "role": "system",
                "content": (
                    "Answer only from the context. Cite the actual policy ID "
                    "shown beside each passage, for example [SEC-114]. "
                    "Use only IDs present in the context. "
                    "If the context does not contain the answer, reply exactly: "
                    "No authorized document contains this answer."
                ),
            },
            {
                "role": "user",
                "content": f"Question: {question}\n\nContext:\n{context}",
            },
        ],
    )
    return response.choices[0].message.content


def main():
    user_id = "alice"
    question = "What does SEC-114 require for laptops?"

    # Step 1: Keyword search is filtered before any evidence reaches the model.
    chunks = retrieve_chunks(question, user_id)

    # Step 2: Generate from authorized evidence only.
    answer = generate_answer(question, chunks)

    print("SIGNED-IN USER\n", user_id)
    print("ACCESS FILTER\n", access_filter(user_id))
    print("QUESTION\n", question)
    print("\nRETRIEVED CHUNKS")
    for chunk in chunks:
        print(f"- [{chunk['policyId']}] {chunk['source']} ({chunk['classification']})")
    print("\nANSWER\n", answer)


if __name__ == "__main__":
    main()

def sign_in(user_name):
    user = USERS[user_name]
    banner = f"### Signed in as {user_name}\n\nAccess: {user['access']}"
    return gr.update(visible=False), gr.update(visible=True), user["id"], banner


def sign_out():
    return gr.update(visible=True), gr.update(visible=False), "", "", "", "", ""


def format_sources(chunks):
    if not chunks:
        return "### Authorized retrieved sources\n\nNone"

    lines = ["### Authorized retrieved sources"]
    for chunk in chunks:
        lines.append(
            f"- [{chunk['policyId']}] {chunk['source']} ({chunk['classification']})"
        )
    return "\n".join(lines)


def run_rag(question, user_id):
    if not question.strip():
        raise gr.Error("Please enter a question.")
    if not user_id:
        raise gr.Error("Please sign in first.")

    chunks = retrieve_chunks(question, user_id)
    answer = generate_answer(question, chunks)
    return (
        f"### Answer\n\n{answer}",
        format_sources(chunks),
        access_filter(user_id),
    )


def build_demo():
    with gr.Blocks(title="Lab 01 - User-Aware Keyword RAG") as demo:
        signed_in_user = gr.State("")

        with gr.Column() as sign_in_panel:
            gr.Markdown(
                "# Lab 01: User-Aware Keyword RAG\n\n"

            )
            user_choice = gr.Dropdown(

                value="Alice - Employee",
                label="Choose a simulated user",
            )
            sign_in_button = gr.Button("Sign in", variant="primary")
            gr.Markdown(
                "**Alice:** Public and Internal documents  \n"
                "**Bob:** Public, Internal, and Confidential documents"
            )

        with gr.Column(visible=False) as rag_panel:
            user_banner = gr.Markdown()
            sign_out_button = gr.Button("Sign out")
            question = gr.Textbox(
                label="Question",
                lines=4,
                placeholder="Ask a policy or troubleshooting question...",
            )
            run_button = gr.Button("Run authorized RAG", variant="primary")
            with gr.Row():
                answer = gr.Markdown()
                sources = gr.Markdown()
            effective_filter = gr.Textbox(
                label="Mandatory Azure AI Search access filter",
                interactive=False,
            )

            with gr.Accordion("Suggested access-test prompts", open=False):
                gr.Examples(
                    examples=[[item] for item in TEST_PROMPTS],
                    inputs=question,
                    label="Click a prompt to copy it into the question box",
                )

        sign_in_button.click(
            sign_in,
            inputs=user_choice,
            outputs=[sign_in_panel, rag_panel, signed_in_user, user_banner],
        )
        sign_out_button.click(
            sign_out,
            outputs=[
                sign_in_panel,
                rag_panel,
                signed_in_user,
                question,
                answer,
                sources,
                effective_filter,
            ],
        )

        run_button.click(
            run_rag,
            inputs=[question, signed_in_user],
            outputs=[answer, sources, effective_filter],
        )
        question.submit(
            run_rag,
            inputs=[question, signed_in_user],
            outputs=[answer, sources, effective_filter],
        )

    return demo
